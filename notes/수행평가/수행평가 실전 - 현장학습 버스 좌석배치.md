## 좌석 표기 & 옆자리 규칙

- 좌석 = `줄_열`. 줄(row) = 앞→뒤 (0줄 맨 앞), 열(col) = 0~3. `seats[row][col]`
- 테스트는 `Bus(5)` (5줄 × 4열)
- 옆자리 쌍: 같은 줄에서 열 (0,1), (2,3). 열 1과 2 사이는 통로 → 옆자리 아님
- `get_partner(row, col)`: 짝수 열이면 오른쪽(+1), 홀수 열이면 왼쪽(-1) (제공)

## 전체 흐름

- 갈 때: `arrange_bus` 1번 → `update_data`로 옆자리 기록
- 올 때: 같은 passengers로 `arrange_bus` 다시 (이번엔 `prev_partner`가 채워져 `is_safe`가 이전 옆자리 막음)
- 올 때 실패하면 `allow_prev_partner=True`로 이 조건만 풀고 재시도

## 📄 전체 제출 코드

```py
import csv
import random


class Passenger:
    def __init__(self, name, gender, fixed_seat, preferred_seat):
        # ▶ TODO 1-1
        self.name = name
        self.gender = gender
        self.fixed_seat = fixed_seat
        self.preferred_seat = preferred_seat
        self.prev_partner = None


class Bus:
    def __init__(self, rows, cols=4, allow_prev_partner=False):
        self.rows = rows
        self.cols = cols
        self.allow_prev_partner = allow_prev_partner
        self.seats = [[None for _ in range(cols)] for _ in range(rows)]

    def get_partner(self, row, col):
        partner_col = col + 1 if col % 2 == 0 else col - 1
        if 0 <= partner_col < self.cols:
            return self.seats[row][partner_col]
        return None

    def display_seats(self):
        for seat_row in self.seats:
            names = []
            # ▶ TODO 1-2
            for p in seat_row:
                names.append("빈자리" if p is None else p.name)
            print(" ".join(names))


def load_passengers_data(file_path):
    def parse_seat(seat_str):
        return tuple(map(int, seat_str.split('_'))) if seat_str else None

    passengers = []
    with open(file_path, newline='', encoding='utf-8-sig') as csvfile:
        reader = csv.DictReader(csvfile)
        for row in reader:
            preferred_seat = {
                "front": int(row['preferred_front']) if row['preferred_front'] else 0,
                "back": int(row['preferred_back']) if row['preferred_back'] else 0
            }
            passenger = Passenger(
                name=row['name'],
                gender=row['gender'],
                fixed_seat=parse_seat(row['fixed_seat']),
                preferred_seat=preferred_seat
            )
            passengers.append(passenger)
    return passengers


def assign_fixed(bus, passengers):
    remaining_passengers = []
    for passenger in passengers:
        if passenger.fixed_seat is not None:
            row, col = passenger.fixed_seat
            if not (0 <= row < bus.rows and 0 <= col < bus.cols):
                raise ValueError(f"[고정석 오류] ({row}, {col}) 은 버스 좌석 범위를 벗어남.")
            # ▶ TODO 1-3
            if bus.seats[row][col] is not None:
                raise ValueError(f"[고정석 오류] ({row}, {col})은 이미 배정됨.")
            # ▶ TODO 1-4
            bus.seats[row][col] = passenger
        else:
            # ▶ TODO 1-5
            remaining_passengers.append(passenger)
    return remaining_passengers


def create_positions(bus):
    positions = []
    for row in range(bus.rows):
        for col in range(bus.cols):
            # ▶ TODO 2-1
            if bus.seats[row][col] is None:
                positions.append((row, col))
    return positions


def can_seat_by_gender(bus, passengers):
    empty_pairs = 0
    half_pairs = {"남": 0, "여": 0}
    for row in range(bus.rows):
        for col in range(0, bus.cols, 2):
            left, right = bus.seats[row][col], bus.seats[row][col + 1]
            if left is None and right is None:
                empty_pairs += 1
            elif left is None or right is None:
                occupant = left if left is not None else right
                half_pairs[occupant.gender] += 1
    need = {"남": 0, "여": 0}
    for passenger in passengers:
        need[passenger.gender] += 1
    # ▶ TODO 2-2
    total = 0
    for g in need:
        rest = max(0, need[g] - half_pairs[g])
        total += (rest + 1) // 2
    return total <= empty_pairs


def is_safe(bus, passenger, row, col):
    partner = bus.get_partner(row, col)
    # ▶ TODO 2-3
    if partner is not None and partner.gender != passenger.gender:
        return False
    if not bus.allow_prev_partner and partner is not None:
        # ▶ TODO 2-4
        if passenger.prev_partner == partner.name or partner.prev_partner == passenger.name:
            return False
    return True


def assign_seats(bus, passengers):
    front_pref, back_pref, non_pref = [], [], []
    for passenger in passengers:
        front, back = passenger.preferred_seat['front'], passenger.preferred_seat['back']
        # ▶ TODO 2-5
        if front > back:
            front_pref.append(passenger)
        elif back > front:
            back_pref.append(passenger)
        else:
            non_pref.append(passenger)
    front_pref.sort(key=lambda p: p.preferred_seat['front'], reverse=True)
    # ▶ TODO 2-6
    back_pref.sort(key=lambda p: p.preferred_seat['back'], reverse=True)

    random.seed(2026)
    random.shuffle(non_pref)
    ordered_passengers = front_pref + back_pref + non_pref

    positions = create_positions(bus)
    if not can_seat_by_gender(bus, ordered_passengers):
        return False
    return assign_seats_backtracking(bus, ordered_passengers, positions)


def assign_seats_backtracking(bus, passengers, positions, index=0):
    if index >= len(passengers):
        return True
    passenger = passengers[index]
    if passenger.preferred_seat["back"] > passenger.preferred_seat["front"]:
        candidate_positions = reversed(positions)
    else:
        candidate_positions = positions
    for row, col in candidate_positions:
        # ▶ TODO 2-7-1
        if bus.seats[row][col] is not None or not is_safe(bus, passenger, row, col):
            continue
        # ▶ TODO 2-7-2 (Choose)
        bus.seats[row][col] = passenger
        # ▶ TODO 2-7-3 (Explore)
        if assign_seats_backtracking(bus, passengers, positions, index + 1):
            return True
        # ▶ TODO 2-8 (Restore)
        bus.seats[row][col] = None
    return False


def update_data(bus):
    for row in range(bus.rows):
        for col in range(bus.cols):
            passenger = bus.seats[row][col]
            if passenger is None:
                continue
            partner = bus.get_partner(row, col)
            # ▶ TODO 3-1
            if partner is not None:
                passenger.prev_partner = partner.name
            else:
                passenger.prev_partner = None


def arrange_bus(rows, passengers, allow_prev_partner=False):
    bus = Bus(rows, allow_prev_partner=allow_prev_partner)
    remaining_passengers = assign_fixed(bus, passengers)
    # ▶ TODO 3-2
    if assign_seats(bus, remaining_passengers):
        return bus
    return None


def main():
    rows = 5
    passengers = load_passengers_data("passengers.csv")

    go_bus = arrange_bus(rows, passengers)
    if go_bus is None:
        print("갈 때 조건을 만족하는 좌석 배치가 없습니다.")
        return
    print("********  갈 때 좌석 배치  ********")
    go_bus.display_seats()

    update_data(go_bus)

    back_bus = arrange_bus(rows, passengers)
    if back_bus is None:
        print("\n올 때 조건을 만족하는 배치가 없어 이전 옆자리 제한을 풀고 다시 시도합니다.")
        back_bus = arrange_bus(rows, passengers, allow_prev_partner=True)
    if back_bus is None:
        print("\n올 때 조건을 만족하는 좌석 배치가 없습니다.")
        return
    print("\n********  올 때 좌석 배치  ********")
    back_bus.display_seats()


if __name__ == "__main__":
    main()
```
