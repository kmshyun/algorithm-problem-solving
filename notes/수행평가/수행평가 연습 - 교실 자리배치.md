# 📝 수행평가 연습: 교실 자리배치 프로그램

> 출처: Notion "2026 프로그래밍과 문제해결"
> 80분 · 3문제(A/B/C) · 100점 · Python 3. 선호도를 반영한 교실 자리배치를 A → B → C 단계로 구현 (성적 미반영 연습). "버스 좌석배치" 실전의 연습 버전.

## 문제 구성

| 문제 | 내용 | 주요 함수 | 배점 |
|------|------|-----------|------|
| A | 학생 정보와 고정석 배치 | `Student.__init__`, `Classroom.display_seats`, `assign_fixed` | 30 |
| B | 조건을 만족하는 자리배치 | 학생 분류·정렬, 빈자리/제약 검사, `assign_seats_backtracking`, `assign_seats` | 55 |
| C | 배치 결과에 따른 학생 정보 갱신 | `update_data`에서 현재 자리·현재 짝 기록 | 15 |

## 규칙 요약

- 좌석은 `행_열` 형식, 행·열 모두 **0부터**. 행 0이 교실 앞쪽.
- 같은 행에서 열 (0,1), (2,3)이 짝. (1,2)는 짝 아님.
- 제약: 선호도(앞/뒤) 반영, **이전 자리(prev_seat)** 와 **이전 짝(prev_partner)** 금지
- 백트래킹: 선택(Choose) → 탐색(Explore) → 복원(Restore)

> 실전(버스)과의 차이: 교실은 제약이 "이전 자리 + 이전 짝", 버스는 "동성 옆자리 + 이전 옆자리". 분류 규칙도 교실은 `front > 0`/`back > 0`, 버스는 `front vs back` 비교.

```py
import csv
import random
from dataclasses import dataclass, field


@dataclass
class Student:
    name: str
    fixed_seat: object = None
    preferred_seat: dict = field(default_factory=lambda: {"front": 0, "back": 0})
    prev_seat: object = None
    prev_partner: object = None


class Classroom:
    def __init__(self, rows, cols, allow_prev_partner=False, allow_prev_seat=False):
        self.rows = rows
        self.cols = cols
        self.allow_prev_partner = allow_prev_partner
        self.allow_prev_seat = allow_prev_seat
        self.seats = [[None for _ in range(cols)] for _ in range(rows)]

    def get_partner(self, row, col):
        partner_col = col + 1 if col % 2 == 0 else col - 1
        if 0 <= partner_col < self.cols:
            return self.seats[row][partner_col]
        return None

    def display_seats(self):
        for seat_row in self.seats:
            print(" ".join(student.name if student else "-" for student in seat_row))


def parse_seat(text):
    if text == "-":
        return None
    row, col = map(int, text.split("_"))
    return row, col


def load_students_data(file_path):
    students = []
    with open(file_path, newline="", encoding="utf-8-sig") as csvfile:
        for row in csv.DictReader(csvfile):
            students.append(Student(
                name=row["name"],
                fixed_seat=parse_seat(row["fixed_seat"] or "-"),
                preferred_seat={
                    "front": int(row["preferred_front"] or 0),
                    "back": int(row["preferred_back"] or 0),
                },
                prev_seat=parse_seat(row["previous_seat"] or "-"),
                prev_partner=row["previous_partner"] or None,
            ))
    return students


def create_positions(classroom):
    positions = []
    for row in range(classroom.rows):
        for col in range(classroom.cols):
            # TODO 4 (5점): 빈자리이면 positions에 추가한다.
            if classroom.seats[row][col] is None:
                positions.append((row, col))
    return positions


def is_safe(classroom, student, row, col):
    if not classroom.allow_prev_seat:
        # TODO 5 (5점): 이전 자리와 현재 후보가 같으면 False를 반환한다.
        if student.prev_seat == (row, col):
            return False
    partner = classroom.get_partner(row, col)
    if not classroom.allow_prev_partner and partner is not None:
        # TODO 6 (5점): 두 학생 중 한쪽이라도 상대를 이전 짝으로 기록하면 False를 반환한다.
        if student.prev_partner == partner.name or partner.prev_partner == student.name:
            return False
    return True


def assign_seats(classroom, students):
    front_pref, back_pref, non_pref = [], [], []
    # TODO 1 (5점): front가 양수면 앞자리 그룹, 아니면 back이 양수면
    # 뒷자리 그룹, 둘 다 양수가 아니면 무선호 그룹에 정확히 한 번 넣는다.
    for student in students:
        if student.preferred_seat['front'] > 0:
            front_pref.append(student)
        elif student.preferred_seat['back'] > 0:
            back_pref.append(student)
        else:
            non_pref.append(student)
    # TODO 2 (5점): 앞자리·뒷자리 그룹을 각각 선호도 내림차순으로 안정 정렬한다.
    front_pref.sort(key=lambda s: s.preferred_seat['front'], reverse=True)
    back_pref.sort(key=lambda s: s.preferred_seat['back'], reverse=True)

    random.seed(2026)
    random.shuffle(non_pref)
    ordered_students = front_pref + back_pref + non_pref
    positions = create_positions(classroom)

    # TODO 3 (5점): 학생 수가 빈자리 수보다 많으면 탐색 없이 False를 반환한다.
    if len(ordered_students) > len(positions):
        return False
    return assign_seats_backtracking(classroom, ordered_students, positions)


def assign_seats_backtracking(classroom, students, positions, index=0):
    if index >= len(students):
        return True
    student = students[index]
    if student.preferred_seat["front"] == 0 and student.preferred_seat["back"] > 0:
        candidate_positions = reversed(positions)
    else:
        candidate_positions = positions
    for row, col in candidate_positions:
        # TODO 7 (5점): 이미 사용 중이거나 안전하지 않은 후보는 건너뛴다.
        if classroom.seats[row][col] is not None or not is_safe(classroom, student, row, col):
            continue

        # TODO 8 (5점, Choose): 현재 학생을 배치한다.
        classroom.seats[row][col] = student

        # TODO 9 (10점, Explore): 다음 학생부터 배치해 성공하면 True를 반환한다.
        if assign_seats_backtracking(classroom, students, positions, index+1):
            return True

        # TODO 10 (5점, Restore): 실패한 현재 배치를 취소한다.
        classroom.seats[row][col] = None
    return False


def read_record(line):
    name, fixed, front, back, previous, partner = line.rstrip("\n").split("|")
    return Student(
        name=name,
        fixed_seat=parse_seat(fixed),
        preferred_seat={"front": int(front), "back": int(back)},
        prev_seat=parse_seat(previous),
        prev_partner=None if partner == "-" else partner,
    )


def main():
    mode = input().strip()
    global create_positions, assign_seats_backtracking
    if mode == "ORDER":
        classroom = Classroom(4, 4)
        students = [read_record(input()) for _ in range(int(input()))]

        # Provided driver observes the ordering independently of search.
        create_positions = lambda c: [(0, 0)] * len(students)

        def show_order(c, ordered, positions):
            print(",".join(student.name for student in ordered))
            return True

        assign_seats_backtracking = show_order
        assign_seats(classroom, students)
        return
    if mode == "POSITIONS":
        rows, cols = map(int, input().split())
        classroom = Classroom(rows, cols)
        for row in range(rows):
            for col, name in enumerate(input().split()):
                if name != "-":
                    classroom.seats[row][col] = Student(name)
        positions = create_positions(classroom)
        print(" ".join(f"{row},{col}" for row, col in positions) or "EMPTY")
        return

    if mode == "SAFE":
        rows, cols, allow_partner, allow_seat = map(int, input().split())
        classroom = Classroom(rows, cols, bool(allow_partner), bool(allow_seat))
        target = read_record(input())
        row, col = map(int, input().split())
        for _ in range(int(input())):
            occupied_row, occupied_col, name, previous_partner = input().split()
            classroom.seats[int(occupied_row)][int(occupied_col)] = Student(
                name,
                prev_partner=None if previous_partner == "-" else previous_partner,
            )
        print(is_safe(classroom, target, row, col))
        return

    if mode == "CAPACITY":
        rows, cols = map(int, input().split())
        classroom = Classroom(rows, cols)
        students = [read_record(input()) for _ in range(int(input()))]

        # The provided driver checks that an impossible count skips search.
        def forbidden_search(*args):
            print("EXPLORED")
            return False

        create_positions = lambda c: [(0, 0)] * (rows * cols)
        assign_seats_backtracking = forbidden_search
        print(assign_seats(classroom, students))
        return

    if mode == "CSV":
        classroom = Classroom(4, 4, False, False)
        students = load_students_data("students.csv")
    elif mode == "ASSIGN":
        rows, cols, allow_partner, allow_seat = map(int, input().split())
        classroom = Classroom(rows, cols, bool(allow_partner), bool(allow_seat))
        students = [read_record(input()) for _ in range(int(input()))]
    else:
        raise ValueError("지원하지 않는 실행 모드입니다.")

    remaining = []
    for student in students:
        if student.fixed_seat is None:
            remaining.append(student)
        else:
            row, col = student.fixed_seat
            classroom.seats[row][col] = student
    success = assign_seats(classroom, remaining)
    if not success and not classroom.allow_prev_partner:
        classroom.allow_prev_partner = True
        success = assign_seats(classroom, remaining)
        print("RELAXED" if success else "FAIL")
    else:
        print("SUCCESS" if success else "FAIL")
    classroom.display_seats()


if __name__ == "__main__":
    main()
```
