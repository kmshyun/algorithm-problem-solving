## 문제 1. 단계별 실험값의 합

```py
def sum_to(n):
    if n == 1:                  # base case
        return 1
    return n + sum_to(n - 1)

n = int(input())
print(sum_to(n))
```

## 문제 2. 중첩 실험 데이터의 측정값 개수

```py
import json

def count_values(data):
    if isinstance(data, int):
        return 1
    total = 0
    for item in data:
        total += count_values(item)
    return total

data = json.loads(input())
print(count_values(data))
```

## 문제 3. 중첩 데이터에서 목표값 찾기

```py
import json

def contains(data, target):
    if isinstance(data, int):
        return data == target
    for item in data:
        if contains(item, target):
            return True
    return False

data = json.loads(input())
target = int(input())
print("YES" if contains(data, target) else "NO")
```

## 문제 5. 대칭 신호 검사 (팰린드롬, 인덱스 방식 O(n))

```py
def is_palindrome(text, left, right):
    if left >= right:
        return True
    if text[left] != text[right]:
        return False
    return is_palindrome(text, left + 1, right - 1)

text = input().strip()
print("YES" if is_palindrome(text, 0, len(text) - 1) else "NO")
```

## 문제 6. 중첩 구조의 최대 깊이

```py
import json

def get_depth(data):
    if isinstance(data, int):
        return 0
    if len(data) == 0:
        return 1
    deepest = 0
    for item in data:
        deepest = max(deepest, get_depth(item))
    return deepest + 1

data = json.loads(input())
print(get_depth(data))
```

## 문제 7. 특정 깊이의 측정값 합계

```py
import json

def sum_at_depth(data, target_depth, cur_depth):
    if isinstance(data, int):
        return data if cur_depth == target_depth else 0
    total = 0
    for item in data:
        total += sum_at_depth(item, target_depth, cur_depth + 1)
```

## 문제 8. 가장 깊은 측정값 분석 (튜플 반환)

```py
import json

def deepest_info(data, depth):
    if isinstance(data, int):
        return depth, data
    max_depth = -1
    total = 0
    for item in data:
        item_depth, item_sum = deepest_info(item, depth + 1)
        if item_depth > max_depth:
            max_depth = item_depth
            total = item_sum
        elif item_depth == max_depth:
            total += item_sum
    return max_depth, total

data = json.loads(input())
max_depth, total = deepest_info(data, 0)
print(max_depth, total)
```

## 문제 12. 합이 S가 되는 조합 (가지치기)

```py
N, M, S = map(int, input().split())

def backtrack(start, picked, cur_sum):
    if cur_sum > S:
        return 0
    if picked == M:
        return 1 if cur_sum == S else 0
    ways = 0
    for num in range(start, N + 1):
        ways += backtrack(num + 1, picked + 1, cur_sum + num)
    return ways

print(backtrack(1, 0, 0))
```

## 문제 13. 조건을 만족하는 암호 만들기

- 첫 자리는 F 금지, 이웃한 두 수는 홀짝이 달라야 함. 없으면 NONE

```py
N, M, F = map(int, input().split())
chosen = []
used = [False] * (N + 1)

def backtrack():
    if len(chosen) == M:
        print(*chosen)
        return 1
    found = 0
    for num in range(1, N + 1):
        if used[num]:
            continue
        if not chosen and num == F:
            continue
        if chosen and chosen[-1] % 2 == num % 2:
            continue
        chosen.append(num)
        used[num] = True
        found += backtrack()
        used[num] = False
        chosen.pop()
    return found

if backtrack() == 0:
    print("NONE")
```

## 문제 14. N×N Queen 배치

```py
N = int(input())
queen_rows = []     # queen_rows[col] = 그 열 퀸의 행

def can_place(row):
    col = len(queen_rows)
    for prev_col, prev_row in enumerate(queen_rows):
        if row == prev_row:
            return False
        if abs(row - prev_row) == abs(col - prev_col):
            return False
    return True

def backtrack():
    if len(queen_rows) == N:
        return 1
    ways = 0
    for row in range(N):
        if not can_place(row):
            continue
        queen_rows.append(row)
        ways += backtrack()
        queen_rows.pop()
    return ways

print(backtrack())
```

## 문제 15. 가능한 발표 시간 배정

```py
N = int(input())
available = []
for _ in range(N):
    line = list(map(int, input().split()))
    available.append(line[1:])          # line[0]은 개수 K

used_time = [False] * (N + 1)

def backtrack(student):
    if student == N:
        return 1
    ways = 0
    for time in available[student]:
        if used_time[time]:
            continue
        used_time[time] = True
        ways += backtrack(student + 1)
        used_time[time] = False
    return ways

print(backtrack(0))
```

## 문제 16. 장애물이 있는 Queen 배치

```py
N, K = map(int, input().split())
blocked = set()
for _ in range(K):
    r, c = map(int, input().split())
    blocked.add((r - 1, c - 1))

queen_rows = []

def can_place(row):
    col = len(queen_rows)
    if (row, col) in blocked:
        return False
    for prev_col, prev_row in enumerate(queen_rows):
        if row == prev_row:
            return False
        if abs(row - prev_row) == abs(col - prev_col):
            return False
    return True

def backtrack():
    if len(queen_rows) == N:
        return 1
    ways = 0
    for row in range(N):
        if not can_place(row):
            continue
        queen_rows.append(row)
        ways += backtrack()
        queen_rows.pop()
    return ways

print(backtrack())
```
