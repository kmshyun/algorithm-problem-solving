## 문제 1. 이동 시간표 만들기

```py
N, M = map(int, input().split())
graph = [[] for _ in range(N + 1)]

for _ in range(M):
    u, v, cost = map(int, input().split())
    graph[u].append((v, cost))
    graph[v].append((u, cost))

for room in range(1, N + 1):
    graph[room].sort()

for room in range(1, N + 1):
    row = [room, len(graph[room])]
    for neighbor, cost in graph[room]:
        row.append(neighbor)
        row.append(cost)
    print(*row)
```

## 문제 2. 긴급 실험 요청 처리 (우선순위 큐)

```py
import heapq

Q = int(input())
heap = []

for _ in range(Q):
    command = input().split()
    if command[0] == "1":
        req_id = int(command[1])
        priority = int(command[2])
        heapq.heappush(heap, (priority, req_id))
    else:
        if heap:
            priority, req_id = heapq.heappop(heap)
            print(req_id)
        else:
            print("EMPTY")
```

- 변수 이름 `id`는 내장 함수를 덮어씀 → `req_id`로

## 문제 3. 가장 빨리 도착하기 (S → T)

```py
import heapq

N, M = map(int, input().split())
S, T = map(int, input().split())
graph = [[] for _ in range(N + 1)]
for _ in range(M):
    u, v, cost = map(int, input().split())
    graph[u].append((v, cost))
    graph[v].append((u, cost))

INF = float("inf")
times = [INF] * (N + 1)
times[S] = 0
heap = [(0, S)]

while heap:
    elapsed, cur = heapq.heappop(heap)
    if elapsed > times[cur]:
        continue
    if cur == T:
        break
    for nxt, cost in graph[cur]:
        new_time = elapsed + cost
        if new_time < times[nxt]:
            times[nxt] = new_time
            heapq.heappush(heap, (new_time, nxt))

print(-1 if times[T] == INF else times[T])
```

## 문제 6. 시료 보관소를 거쳐 가기 (경유지)

```py
import heapq

N, M = map(int, input().split())
S, K, T = map(int, input().split())
graph = [[] for _ in range(N + 1)]
for _ in range(M):
    u, v, cost = map(int, input().split())
    graph[u].append((v, cost))
    graph[v].append((u, cost))

INF = float("inf")

def dijkstra(start):
    times = [INF] * (N + 1)
    times[start] = 0
    heap = [(0, start)]
    while heap:
        elapsed, cur = heapq.heappop(heap)
        if elapsed > times[cur]:
            continue
        for nxt, cost in graph[cur]:
            new_time = elapsed + cost
            if new_time < times[nxt]:
                times[nxt] = new_time
                heapq.heappush(heap, (new_time, nxt))
    return times

from_start = dijkstra(S)
from_storage = dijkstra(K)

if from_start[K] == INF or from_storage[T] == INF:
    print(-1)
else:
    print(from_start[K] + from_storage[T])
```

## 문제 7. 점검하고 돌아오기 (왕복, 간선 뒤집기)

```py
import heapq

N, M = map(int, input().split())
S, L = map(int, input().split())
graph = [[] for _ in range(N + 1)]
reverse_graph = [[] for _ in range(N + 1)]
for _ in range(M):
    u, v, cost = map(int, input().split())
    graph[u].append((v, cost))
    reverse_graph[v].append((u, cost))      # 방향 뒤집어 저장

INF = float("inf")

def dijkstra(start, adj):
    times = [INF] * (N + 1)
    times[start] = 0
    heap = [(0, start)]
    while heap:
        elapsed, cur = heapq.heappop(heap)
        if elapsed > times[cur]:
            continue
        for nxt, cost in adj[cur]:
            new_time = elapsed + cost
            if new_time < times[nxt]:
                times[nxt] = new_time
                heapq.heappush(heap, (new_time, nxt))
    return times

go = dijkstra(S, graph)                 # S → 각 구역
back = dijkstra(S, reverse_graph)       # 각 구역 → S

count = 0
for room in range(1, N + 1):
    if go[room] + back[room] <= L:
        count += 1
print(count)
```

## 문제 8. 탐사 로봇의 에너지 (격자 다익스트라)

```py
import heapq

R, C = map(int, input().split())
energy = [list(map(int, input().split())) for _ in range(R)]

INF = float("inf")
cost = [[INF] * C for _ in range(R)]
cost[0][0] = 0                          # 출발 칸 에너지는 안 씀
heap = [(0, 0, 0)]                      # (누적 에너지, r, c)

while heap:
    used, r, c = heapq.heappop(heap)
    if used > cost[r][c]:
        continue
    for nr, nc in [(r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)]:
        if 0 <= nr < R and 0 <= nc < C:
            new_used = used + energy[nr][nc]
            if new_used < cost[nr][nc]:
                cost[nr][nc] = new_used
                heapq.heappush(heap, (new_used, nr, nc))

print(cost[R - 1][C - 1])
```
