```py
import json

def get_depth(data):
    if isinstance(data, int):
        return 0
    if len(data) == 0:
        return 1
    max_depth = 0
    for item in data:
        max_depth = max(max_depth, get_depth(item))
    return max_depth + 1

data = json.loads(input())
print(get_depth(data))
```

## 문제 6. 중복 없는 M개 선택 (백트래킹, 순열)

```py
N, M = map(int, input().split())
state = []
used = [False] * (N + 1)

def backtrack():
    if len(state) == M:
        print(*state)
        return
    for i in range(1, N + 1):
        if used[i]:
            continue
        state.append(i)      # Choose
        used[i] = True
        backtrack()          # Explore
        state.pop()          # Restore
        used[i] = False

backtrack()
```

## 문제 7. 조건을 만족하는 암호 만들기 (백트래킹, 가지치기)

```py
N, M, F = map(int, input().split())
state = []
used = [False] * (N + 1)

def backtrack():
    if len(state) == M:
        print(*state)
        return 1
    count = 0
    for i in range(1, N + 1):
        if used[i]:
            continue
        if not state and i == F:
            continue
        if state and state[-1] % 2 == i % 2:
            continue
        state.append(i)
        used[i] = True
        count += backtrack()
        used[i] = False
        state.pop()
    return count

if backtrack() == 0:
    print("NONE")
```

## 문제 11. 각 구역까지 몇 번? (BFS)

```py
from collections import deque

N, M = map(int, input().split())
vertices = input().split()

graph = {v: [] for v in vertices}
for _ in range(M):
    u, v = input().split()
    graph[u].append(v)
    graph[v].append(u)

S = input().strip()

distance = {v: -1 for v in vertices}
distance[S] = 0
queue = deque([S])

while queue:
    v = queue.popleft()
    for w in graph[v]:
        if distance[w] != -1:
            continue
        distance[w] = distance[v] + 1
        queue.append(w)

for v in sorted(vertices):
    print(f"{v}: {distance[v]}")
```

## 문제 12. 가장 짧은 점검 경로 (BFS, 경로 복원)

```py
from collections import deque

N, M = map(int, input().split())
vertices = input().split()

graph = {v: [] for v in vertices}
for _ in range(M):
    u, v = input().split()
    graph[u].append(v)
    graph[v].append(u)
for v in vertices:
    graph[v].sort()

S, T = input().split()

distance = {v: -1 for v in vertices}
parent = {v: None for v in vertices}
distance[S] = 0
queue = deque([S])

while queue:
    v = queue.popleft()
    if v == T:
        break
    for w in graph[v]:
        if distance[w] != -1:
            continue
        distance[w] = distance[v] + 1
        parent[w] = v
        queue.append(w)

if distance[T] == -1:
    print(-1)
    print("NONE")
else:
    path = []
    cur = T
    while cur is not None:
        path.append(cur)
        cur = parent[cur]
    path.reverse()
    print(distance[T])
    print(*path)
```

## 문제 13. DFS와 BFS가 찾은 구역

N, M = map(int, input().split())
vertices = input().split()

    for w in graph[v]:
        if w not in dfs_visited:

## 문제 14. 가장 빨리 도착하기 (다익스트라)

```py
import heapq

N, M = map(int, input().split())
S, T = map(int, input().split())

graph = [[] for _ in range(N + 1)]
for _ in range(M):
    u, v, cost = map(int, input().split())
    graph[u].append((v, cost))
    graph[v].append((u, cost))

INF = float("inf")
times = [INF] * (N + 1)
times[S] = 0
heap = [(0, S)]

while heap:
    elapsed, cur = heapq.heappop(heap)
    if elapsed > times[cur]:
        continue
    if cur == T:
        break
    for nxt, cost in graph[cur]:
        new_time = elapsed + cost
        if new_time < times[nxt]:
            times[nxt] = new_time
            heapq.heappush(heap, (new_time, nxt))

if times[T] == INF:
    print(-1)
else:
    print(times[T])
```

## 문제 15. 시료 보관소를 거쳐 가기 (다익스트라, 경유지)

```py
import heapq

N, M = map(int, input().split())
S, K, T = map(int, input().split())

graph = [[] for _ in range(N + 1)]
for _ in range(M):
    u, v, cost = map(int, input().split())
    graph[u].append((v, cost))
    graph[v].append((u, cost))

def dijkstra(start):
    INF = float("inf")
    times = [INF] * (N + 1)
    times[start] = 0
    heap = [(0, start)]
    while heap:
        elapsed, cur = heapq.heappop(heap)
        if elapsed > times[cur]:
            continue
        for nxt, cost in graph[cur]:
            new_time = elapsed + cost
            if new_time < times[nxt]:
                times[nxt] = new_time
                heapq.heappush(heap, (new_time, nxt))
    return times

from_s = dijkstra(S)
from_k = dijkstra(K)

if from_s[K] == float("inf") or from_k[T] == float("inf"):
    print(-1)
else:
    print(from_s[K] + from_k[T])
```
