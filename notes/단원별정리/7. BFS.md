# 1. BFS란

- Breadth-First Search (너비 우선 탐색): 시작 정점에서 가까운 정점부터 차례로 방문
- 거리 1 전부 → 거리 2 전부 → … (연못에 돌 던지면 물결이 퍼지는 것)
- DFS가 "한 길로 끝까지"라면 BFS는 "한 겹씩 넓게"
- 큐(FIFO)를 쓰는 이유: 먼저 발견한 걸 먼저 처리해야 가까운 정점부터 방문. `collections.deque`의 `popleft()`는 O(1) (리스트 `pop(0)`은 O(n))

# 2. 기본 코드

```py
from collections import deque

n = int(input())
k = int(input())
g = [[] for _ in range(n + 1)]

for i in range(k):
    a, b = map(int, input().split())
    g[a].append(b)
    g[b].append(a)

visited = [False] * (n + 1)

def bfs(s): # BFS는 재귀가 아니다!
    q = deque([s])
    visited[s] = True

    while q:
        cur = q.popleft()
        for i in g[cur]:
            if not visited[i]:
                visited[i] = True  # 큐에 넣을 때 반드시 방문 처리
                q.append(i)
```

## 시간 복잡도 / 주의

- 인접 리스트: O(V + E) / 인접 행렬: O(V²)
- 넓은 그래프는 BFS의 큐, 깊은 그래프는 DFS의 재귀 깊이를 조심
- `list.pop(0)`은 O(N) → 반드시 `deque.popleft()`
- 격자에서 범위 검사를 무조건 먼저! (short-circuit)

# 3. 2차원 배열(격자)에서의 BFS

## 기본 코드: 미로 최단 거리

```py
from collections import deque
import sys
input = sys.stdin.readline

R, C = map(int, input().split())
a = [list(map(int, input().strip())) for _ in range(R)]   # 11011처럼 붙어서 들어오는 입력
dist = [[-1] * C for _ in range(R)]                        # -1 = 아직 안 감

dx = [-1, 1, 0, 0]   # 상 하 좌 우
dy = [0, 0, -1, 1]

q = deque([(0, 0)])
dist[0][0] = 1       # 시작 칸도 1칸으로 셈

while q:
    x, y = q.popleft()
    for d in range(4):
        nx, ny = x + dx[d], y + dy[d]
        if 0 <= nx < R and 0 <= ny < C and dist[nx][ny] == -1 and a[nx][ny] == 1:
            dist[nx][ny] = dist[x][y] + 1
            q.append((nx, ny))

print(dist[R-1][C-1])   # 못 가면 -1
```

## 응용 ① 섬 개수·크기를 BFS로

```py
from collections import deque
import sys
input = sys.stdin.readline

R, C = map(int, input().split())
a = [list(map(int, input().split())) for _ in range(R)]
visited = [[False] * C for _ in range(R)]

dx = [-1, 1, 0, 0]
dy = [0, 0, -1, 1]

def bfs(sx, sy):
    q = deque([(sx, sy)])
    visited[sx][sy] = True
    cnt = 1
    while q:
        x, y = q.popleft()
        for d in range(4):
            nx, ny = x + dx[d], y + dy[d]
            if 0 <= nx < R and 0 <= ny < C and not visited[nx][ny] and a[nx][ny] == 1:
                visited[nx][ny] = True
                q.append((nx, ny))
                cnt += 1
    return cnt

s = []
for i in range(R):
    for j in range(C):
        if a[i][j] == 1 and not visited[i][j]:
            s.append(bfs(i, j))

print(len(s))
print(*sorted(s))
```

## 응용 ② 멀티소스 BFS (여러 시작점 동시에)

```py
from collections import deque
import sys
input = sys.stdin.readline

R, C = map(int, input().split())
a = [list(map(int, input().split())) for _ in range(R)]   # 1: 불, 0: 빈칸, -1: 벽
dist = [[-1] * C for _ in range(R)]

dx = [-1, 1, 0, 0]
dy = [0, 0, -1, 1]

q = deque()
for i in range(R):
    for j in range(C):
        if a[i][j] == 1:          # 시작점을 전부 큐에
            q.append((i, j))
            dist[i][j] = 0

while q:
    x, y = q.popleft()
    for d in range(4):
        nx, ny = x + dx[d], y + dy[d]
        if 0 <= nx < R and 0 <= ny < C and dist[nx][ny] == -1 and a[nx][ny] == 0:
            dist[nx][ny] = dist[x][y] + 1
            q.append((nx, ny))

ans = 0
for i in range(R):
    for j in range(C):
        if a[i][j] == 0:
            if dist[i][j] == -1:
                print(-1)
                sys.exit()
            ans = max(ans, dist[i][j])

print(ans)
```

- 방문 처리(dist 기록)는 큐에 넣을 때! 꺼낼 때 하면 같은 칸이 여러 번 들어가 시간/메모리 초과

# 4. DFS vs BFS

- BFS: 가중치 없는 그래프에서 최단 경로 보장 (처음 발견 = 최단 거리). DFS: 보장 X
- 가중치가 다르면 BFS도 최단 보장 X → 0/1만이면 0-1 BFS, 양수면 다익스트라, 음수 포함이면 벨만-포드
- DFS: 스택(재귀) / BFS: 큐(deque)
- DFS 주로: 연결 요소, 경로 존재, 모든 경우 / BFS 주로: 최단 거리, 최소 이동, 미로 최단

# 5. BFS의 활용

## 활용 ① 최단거리 구하기

```py
from collections import deque
import sys
input = sys.stdin.readline

a, b = map(int, input().split())
g = [[] for _ in range(a + 1)]

for _ in range(b):
    k, l = map(int, input().split())
    g[k].append(l)
    g[l].append(k)
    
s, t = map(int, input().split())
dist = [-1] * (a+1)
dist[s] = 0
q = deque([s])

while q:
    v = q.popleft()
    if v == t:
        break
    for w in g[v]:
        if dist[w] != -1:
            continue
        q.append(w)
        dist[w] = dist[v] + 1

print(dist[t])
```

## 활용 ③ 최단거리 복원 (부모 포인터)

```py
from collections import deque
import sys
input = sys.stdin.readline

a, b = map(int, input().split())
g = [[] for _ in range(a + 1)]

for _ in range(b):

    g[l].append(v)

    if v == t:
        break

            q.append(w)
```

## 활용 ④ 미로찾기 (경로 복원)

```py
from collections import deque
import sys
input = sys.stdin.readline

a, b = map(int, input().split())
g = [input().rstrip() for _ in range(a)]

sr, sc, tr, tc = map(int, input().split())
sr -= 1; sc -= 1; tr -= 1; tc -= 1

d = [[-1] * b for _ in range(a)]
p = [[None] * b for _ in range(a)]

d[sr][sc] = 0
q = deque([(sr, sc)])

while q:
    r, c = q.popleft()
    if r == tr and c == tc:
        break
    for nr, nc in [(r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)]:
        if 0 <= nr < a and 0 <= nc < b and g[nr][nc] == '0' and d[nr][nc] == -1:
            q.append((nr, nc))
            d[nr][nc] = d[r][c] + 1
            p[nr][nc] = (r, c)

print(d[tr][tc])

if d[tr][tc] != -1:
    path = []
    curr = (tr, tc)
    while curr is not None:
        path.append(curr)
        curr = p[curr[0]][curr[1]]
    path.reverse()
    print(path)
```

## 활용 ⑤ 거리가 정확히 k인 정점 모두 찾기

```py
from collections import deque
import sys
input = sys.stdin.readline

a, b, k, s = map(int, input().split())
g = [[] for _ in range(a + 1)]

for _ in range(b):
    v, l = map(int, input().split())
    g[v].append(l)
    g[l].append(v)

dist = [-1] * (a + 1)
dist[s] = 0
q = deque([s])

while q:
    v = q.popleft()
    if dist[v] == k:   # 거리 k까지만 필요 → 가지치기
        continue
    for w in g[v]:
        if dist[w] == -1:
            dist[w] = dist[v] + 1
            q.append(w)

r = [i for i in range(1, a + 1) if dist[i] == k]

if r:
    print(*r)
else:
    print(-1)
```

## 활용 ⑥ 위상 정렬 (Kahn 알고리즘)

```py
import heapq
import sys
input = sys.stdin.readline

v, n = map(int, input().split())
adj = [[] for _ in range(v + 1)]
deg = [0] * (v + 1)

for _ in range(n):
    a, b = map(int, input().split())
    adj[a].append(b)
    deg[b] += 1

pq = [i for i in range(1, v + 1) if deg[i] == 0]
heapq.heapify(pq)
res = []

while pq:
    x = heapq.heappop(pq)
    res.append(x)
    for nx in adj[x]:
        deg[nx] -= 1
        if deg[nx] == 0:
            heapq.heappush(pq, nx)

if len(res) == v:
    for i in res:
        print(i)
else:
    print(-1)   # 사이클 있음
```

- 진입차수(indegree) 0 = 지금 할 수 있는 일. 힙을 쓰면 번호 작은 것부터(사전순)
- `len(res) != v`면 사이클

## 예제 1. 순간이동 (상태 BFS)

```py
from collections import deque
import sys
input = sys.stdin.readline

N, K = map(int, input().split())

MAX = 100000
dist = [-1] * (MAX + 1)

q = deque([N])
dist[N] = 0

while q:
    curr = q.popleft()
    if curr == K:
        print(dist[curr])
        break
    for nxt in (curr - 1, curr + 1, curr * 2):
        if 0 <= nxt <= MAX and dist[nxt] == -1:
            dist[nxt] = dist[curr] + 1
            q.append(nxt)
```

## 예제 2. 촌수 계산

```py
from collections import deque

n = int(input())
a, b = map(int, input().split())
k = int(input())

g = [[] for _ in range(n+1)]
for i in range(k):
    x, y = map(int, input().split())
    g[x].append(y)
    g[y].append(x)

visited = [False] * (n+1)
distance = [0] * (n+1)
q = deque([a])

def bfs(a, b):
    visited[a] = True
    while q:
        c = q.popleft()
        if c == b:
            return distance[b]
        for i in g[c]:
            if not visited[i]:
                visited[i] = True
                distance[i] = distance[c] + 1
                q.append(i)
    return -1

print(bfs(a, b))
```

## 예제 4. 토마토 (2차원, 멀티소스)

```py
from collections import deque
import sys
input = sys.stdin.readline

n, m = map(int, input().split())
g = [list(map(int, input().split())) for _ in range(m)]

q = deque()
for i in range(m):
    for j in range(n):
        if g[i][j] == 1:
            q.append((i, j))

dx = [-1, 1, 0, 0]
dy = [0, 0, -1, 1]

while q:
    x, y = q.popleft()
    for i in range(4):
        nx = x + dx[i]
        ny = y + dy[i]
        if 0 <= nx < m and 0 <= ny < n:
            if g[nx][ny] == 0:
                g[nx][ny] = g[x][y] + 1
                q.append((nx, ny))

ans = 0
for row in g:
    for val in row:
        if val == 0:
            print(-1)
            exit(0)
    ans = max(ans, max(row))

print(ans-1)
```

- 입력은 `n m`(가로, 세로)인데 배열은 `g[세로][가로]` → `range(m)`으로 읽음

## 예제 5. 토마토 (3차원, 6방향)

```py
from collections import deque
import sys
input = sys.stdin.readline

n, m, h = map(int, input().split())
g = []
for _ in range(h):
    layer = [list(map(int, input().split())) for _ in range(m)]
    g.append(layer)

q = deque()
for z in range(h):
    for y in range(m):
        for x in range(n):
            if g[z][y][x] == 1:
                q.append((z, y, x))

dx = [0, 0, 1, -1, 0, 0]
dy = [1, 0, 0, 0, 0, -1]
dz = [0, 1, 0, 0, -1, 0]

while q:
    z, y, x = q.popleft()
    for i in range(6):
        nx = x + dx[i]
        ny = y + dy[i]
        nz = z + dz[i]
        if 0 <= nx < n and 0 <= ny < m and 0 <= nz < h:
            if g[nz][ny][nx] == 0:
                g[nz][ny][nx] = g[z][y][x] + 1
                q.append((nz, ny, nx))

ans = 0
for layer in g:
    for row in layer:
        for val in row:
            if val == 0:
                print(-1)
                exit(0)
            ans = max(ans, val)

print(ans-1)
```

## 예제 8. 나이트의 최소 이동

```py
from collections import deque
n = int(input())
g = [[0 for _ in range(n)] for _ in range(n)]

dr = [2, 2, -2, -2, -1, -1, 1, 1]
dc = [1, -1, 1, -1, 2, -2, 2, -2]

sr, sc = map(int, input().split())
sr, sc = sr - 1, sc - 1
tr, tc = map(int, input().split())
tr, tc = tr - 1, tc - 1

visited = [[False] * (n) for _ in range(n)]
visited[sr][sc] = True
q = deque([(sr, sc)])

while q:
    r, c = q.popleft()
    if r == tr and c == tc:
        break
    for i in range(8):
        nr = r + dr[i]
        nc = c + dc[i]
        if 0 <= nr < n and 0 <= nc < n:
            if not visited[nr][nc]:
                g[nr][nc] = g[r][c] + 1
                visited[nr][nc] = True
                q.append((nr, nc))

print(g[tr][tc])
```

## 예제 10. 0과 1로만 이루어진 배수 (나머지로 방문 체크)

```py
from collections import deque
import sys
input = sys.stdin.readline

n = int(input())
ULL_MAX = (1 << 64) - 1

visited = {1 % n}           # 나온 나머지들 (set)
q = deque([1])
ans = 0

while q:
    cur = q.popleft()
    if cur % n == 0:
        ans = cur
        break
    for i in (0, 1):        # 0을 먼저 붙여야 작은 수부터
        nxt = cur * 10 + i
        if nxt > ULL_MAX:
            continue
        if nxt % n not in visited:
            visited.add(nxt % n)
            q.append(nxt)

print(ans)
```

- 핵심: 상태를 "숫자 자체"가 아니라 "n으로 나눈 나머지"로 잡음. n이 크면 `[False]*n` 리스트는 메모리 초과 → set

# ⚠️ n차원 배열 읽는 순서

- 규칙 1: 인덱스 순서 = 입력 for문 순서 (바깥→안쪽). 입력 첫 줄 `n(가로) m(세로) h(높이)` ↔ 인덱스 `g[z][y][x]` (정반대!)
- 규칙 2: 만들 때 가장 안쪽 `* 크기`가 마지막 인덱스
- 규칙 3: 범위 검사는 인덱스 자리↔크기 짝 맞추기. 정사각형 예제는 n/m 바꿔도 통과 → 가로≠세로 입력으로 검산
- 규칙 4: x/y 이름은 역할로 고정 (헷갈리면 r/c/z)
- 규칙 5: 방향 배열은 두 개씩 짝지어
